		This session builds the language of closeness in its two required dialects. The \emph{topological}
		dialect: spaces, bases, and the three constructions (subspace, product, \textbf{quotient}) that

		profinite compactness in S34, $\Z_p$ in S19) and \textbf{completion} (adding exactly the points Cauchy
		sequences deserved: $\Q\to\R$ here, $\Q\to\Q_p$ in S17--S18, $k((t))$ in S21). All main proofs are
		complete; the Zariski contrast (S1) and the adele preview (S22) are woven in throughout.
	\end{abstract}

	\section{Topological Spaces, Constructions, and Continuity}
	% ══════════════════════════════════════════════
	
	\begin{definition}[Topology]
		A \textbf{topology} on a set $X$ is a collection $\tau\subseteq\Pow(X)$ of \textbf{open sets} with:
		(T1) $\varnothing,X\in\tau$; (T2) arbitrary unions of members of $\tau$ lie in $\tau$;
		(T3) finite intersections of members of $\tau$ lie in $\tau$. A set is \textbf{closed} if its
		complement is open. The pair $(X,\tau)$ is a \textbf{topological space}.
	\end{definition}
	
	\begin{example}[First examples]\label{ex:topologies}
		\begin{itemize}
			\item \emph{Discrete:} $\tau=\Pow(X)$; every subset open (and closed). \emph{Indiscrete:} $\tau=\{\varnothing,X\}$.
			\item \emph{Standard on $\R^n$:} generated by open balls $B_\varepsilon(x)$ (made precise in \S2).
			\item \emph{Cofinite:} $\tau=\{\varnothing\}\cup\{U:X\setminus U\text{ finite}\}$ on infinite $X$.
			\item \emph{Zariski:} on $\Spec R$ (S1), closed sets $V(I)=\{\p\in\Spec R:I\subseteq\p\}$; not Hausdorff --- Remark~\ref{rem:zariski}.
		\end{itemize}
	\end{example}
	
	\begin{definition}[Basis]
		A \textbf{basis} for a topology on $X$ is $\mathcal B\subseteq\Pow(X)$ with (B1) $\bigcup_{B\in\mathcal B}B=X$;
		(B2) if $x\in B_1\cap B_2$ with $B_i\in\mathcal B$, then $\exists B_3\in\mathcal B$ with $x\in B_3\subseteq B_1\cap B_2$.
	\end{definition}
	
	\begin{proposition}[A basis generates a topology]\label{prop:basis}
		If $\mathcal B$ satisfies (B1),(B2), then $\tau=\{U\subseteq X:\forall x\in U\ \exists B\in\mathcal B,\ x\in B\subseteq U\}$
		is a topology, and $\tau$ is exactly the set of unions of members of $\mathcal B$.
	\end{proposition}
	\begin{proof}
		$\varnothing\in\tau$ vacuously; $X\in\tau$ by (B1). Unions: a witness for $U_j$ witnesses $\bigcup_i U_i$.
		Intersections: witnesses $B_1,B_2$ for $x\in U_1\cap U_2$; (B2) gives $B_3$ with $x\in B_3\subseteq U_1\cap U_2$.
		Every union of basis elements is in $\tau$; conversely $U=\bigcup\{B\in\mathcal B:B\subseteq U\}$.
	\end{proof}
	
	\begin{definition}[Subspace topology]
		For $Y\subseteq X$, opens of the \textbf{subspace topology} are exactly $\{U\cap Y:U\in\tau_X\}$.
	\end{definition}
	
	\begin{example}\label{ex:subspace}
		In $Y=(0,1]\subseteq\R$ the set $(\tfrac12,1]$ is open (it is $(\tfrac12,2)\cap Y$) though not open in $\R$.
		In $Y=\Z\subseteq\R$ every subset is open, since $\{n\}=(n-\tfrac12,n+\tfrac12)\cap\Z$.
	\end{example}
	
	\begin{definition}[Product topology]
		On $X\times Y$, basis $\{U\times V:U\in\tau_X,\ V\in\tau_Y\}$. For a family $(X_i)_{i\in I}$, basis
		$\prod_i U_i$ with $U_i$ open and $U_i=X_i$ for all but finitely many $i$.
	\end{definition}
	
	\begin{remark}[Infinite products, and a preview of adeles]\label{rem:adeles}
		The ``all but finitely many'' restriction defines the product topology; the \textbf{box topology}
		drops it and is too fine (products of compact spaces fail to be compact, etc.). The same discipline
		returns in S22: the adele ring $A_\Q$ is a \emph{restricted product},
		$\{(x_v)_v:x_p\in\Z_p\text{ for all but finitely many }p\}$ --- controlled infinity, now arithmetic.
	\end{remark}
	
	\begin{definition}[Quotient topology]
		For a surjection $q:X\to Y$, the \textbf{quotient topology} on $Y$ is
		$\tau_Y=\{V\subseteq Y:q^{-1}(V)\in\tau_X\}$: the finest topology making $q$ continuous.
	\end{definition}
	
	\begin{example}[Quotients everywhere]\label{ex:quotients}
		\begin{itemize}
			\item $[0,1]/\{0\sim1\}\cong S^1$: identify endpoints of an interval.
			\item $\R/\Z\cong S^1$ via $[x]\mapsto e^{2\pi ix}$.
			\item $D^2/\partial D^2\cong S^2$: collapse the boundary of a disk to a point.
			\item $\Spec(R/I)$ inside $\Spec R$ (S1, S28): primes containing $I$.
		\end{itemize}
	\end{example}
	
	\begin{definition}[Continuity; homeomorphism]
		$f:X\to Y$ is \textbf{continuous} if $f^{-1}(V)$ is open for every open $V\subseteq Y$. A
		\textbf{homeomorphism} is a bijective continuous map with continuous inverse; then $X\cong Y$ topologically.
	\end{definition}
	
	\begin{proposition}\label{prop:cont-comp}
		Compositions and identities of continuous maps are continuous; for a bijection $f$, being a
		homeomorphism $\iff$ $f^{-1}(V)$ open for all opens $V$ $\iff$ $f(U)$ open for all opens $U$.
	\end{proposition}
	
	\begin{proposition}[Compact $\to$ Hausdorff bijection is a homeomorphism]\label{prop:compact-hausdorff}
		If $X$ is compact, $Y$ is Hausdorff, and $f:X\to Y$ is a continuous bijection, then $f$ is a homeomorphism.
	\end{proposition}
	\begin{proof}
		It suffices that $f$ be closed. Let $C\subseteq X$ be closed; then $C$ is compact
		(Proposition~\ref{prop:closed-in-compact}); $f(C)$ is compact (Proposition~\ref{prop:compact-image});
		and $f(C)$ is closed in $Y$ (Proposition~\ref{prop:compact-in-hausdorff}). Hence $f$ is closed.
	\end{proof}
	
	\begin{example}\label{ex:homeo}
		$\tan$ gives $(0,1)\cong\R$ (rescaled). The map $[0,1)\to S^1$, $t\mapsto e^{2\pi it}$, is continuous
		and bijective but \emph{not} a homeomorphism (Corollary~\ref{cor:compact-invariant} explains why cleanly).
	\end{example}
	
	\begin{definition}[Hausdorff]
		$X$ is \textbf{Hausdorff} (T2) if distinct points have disjoint open neighbourhoods.
	\end{definition}
	
	\begin{proposition}[Hausdorff consequences]\label{prop:hausdorff}
		(a) Limits of sequences are unique in Hausdorff spaces. (b) Compact subsets of Hausdorff spaces are closed.
	\end{proposition}
	\begin{proof}
		(a) Disjoint $U\ni x$, $V\ni y$ force $x_n\in U\cap V=\varnothing$ for large $n$. (b) For $x\notin K$,
		each $y\in K$ has disjoint $U_y\ni x$, $V_y\ni y$; finitely many $V_{y_i}$ cover $K$; then
		$U=\bigcap_i U_{y_i}$ is an open neighbourhood of $x$ disjoint from $K$.
	\end{proof}
	
	\begin{remark}[Zariski is not Hausdorff]\label{rem:zariski}
		In $\Spec\Z$ (S1) any two nonempty opens meet: $D(m)\cap D(n)=D(mn)\neq\varnothing$. The Zariski
		topology sacrifices Hausdorffness to remember algebraic specialization; metric and \'etale topologies
		(S17--, S52) restore separation where needed.
	\end{remark}

	\section{Metric Spaces: Balls, Cauchy, Completeness}
	% ══════════════════════════════════════════════
	
	\begin{definition}[Metric]
		A \textbf{metric} on $X$ is $d:X\times X\to\R_{\ge0}$ with $d(x,y)=0\iff x=y$; symmetry; and the
		triangle inequality $d(x,z)\le d(x,y)+d(y,z)$.
	\end{definition}
	
	\begin{example}\label{ex:metrics}
		\begin{itemize}
			\item Euclidean $d_2$, taxicab $d_1$, max $d_\infty$ on $\R^n$ (all give the standard topology).
			\item Discrete metric $d(x,y)=1$ for $x\neq y$; induces the discrete topology.
			\item Sup metric on $C[0,1]$: $d_{\sup}(f,g)=\sup_{t}|f(t)-g(t)|$; convergence $=$ uniform convergence.
			\item \emph{Preview (S17):} $p$-adic metric $d_p(x,y)=p^{-v_p(x-y)}$ on $\Q$, with the ultrametric
			inequality $d_p(x,z)\le\max\{d_p(x,y),d_p(y,z)\}$ --- a geometry where all triangles are isosceles (S19).
		\end{itemize}
	\end{example}
	
	\begin{proposition}[Balls form a basis]\label{prop:balls}
		The open balls $B(x,r)$ satisfy (B1),(B2); hence every metric space carries a canonical metric topology.
	\end{proposition}
	\begin{proof}
		(B1): $x\in B(x,1)$. (B2): if $z\in B(x,r)\cap B(y,s)$ set $\varepsilon=\min\{r-d(x,z),\,s-d(y,z)\}>0$;
		then for $w\in B(z,\varepsilon)$: $d(x,w)\le d(x,z)+d(z,w)<r$, so $B(z,\varepsilon)\subseteq B(x,r)$; symmetrically for $B(y,s)$.
	\end{proof}
	
	\begin{proposition}\label{prop:metric-hausdorff}
		Every metric space is Hausdorff; hence limits of sequences are unique.
	\end{proposition}
	\begin{proof}
		For $x\neq y$, $r=d(x,y)/2$ gives disjoint $B(x,r),B(y,r)$ by the triangle inequality; apply Proposition~\ref{prop:hausdorff}(a).
	\end{proof}
	
	\begin{definition}[Convergence; Cauchy; complete]
		$x_n\to x$ means $\forall\varepsilon>0\ \exists N\ \forall n\ge N:\ d(x_n,x)<\varepsilon$. $(x_n)$ is
		\textbf{Cauchy} if $\forall\varepsilon>0\ \exists N\ \forall m,n\ge N:\ d(x_m,x_n)<\varepsilon$.
		$(X,d)$ is \textbf{complete} if every Cauchy sequence converges in $X$.
	\end{definition}
	
	\begin{proposition}\label{prop:conv-cauchy}
		Every convergent sequence is Cauchy.
	\end{proposition}
	\begin{proof}
		Given $\varepsilon$, choose $N$ with $d(x_n,x)<\varepsilon/2$ for $n\ge N$; then
		$d(x_m,x_n)\le d(x_m,x)+d(x,x_n)<\varepsilon$.
	\end{proof}
	
	\begin{example}[Incompleteness as missing points]\label{ex:incomplete}
		In $\Q$: the decimals $1,\,1.4,\,1.41,\,1.414,\dots$ of $\sqrt2$ are Cauchy (they converge in $\R$;
		Proposition~\ref{prop:conv-cauchy}) but have no limit in $\Q$. In $(0,1)$: $x_n=1/n$ is Cauchy with no
		limit in the space. In both cases a point is missing (Figure~1).
	\end{example}
	
	\begin{theorem}[Completeness of the sup metric]\label{thm:sup-complete}
		$(C[0,1],d_{\sup})$ is complete: every Cauchy sequence of continuous functions converges uniformly to a continuous function.
	\end{theorem}
	\begin{proof}
		Let $(f_n)$ be Cauchy. For each $t$, $(f_n(t))$ is Cauchy in $\R$ (since $|f_m(t)-f_n(t)|\le d_{\sup}(f_m,f_n)$),
		hence converges; define $f(t)=\lim_n f_n(t)$. \emph{Uniformity:} given $\varepsilon$, choose $N$ with
		$d_{\sup}(f_m,f_n)<\varepsilon$ for $m,n\ge N$; fixing $n\ge N$ and letting $m\to\infty$ gives
		$|f(t)-f_n(t)|\le\varepsilon$ for all $t$: $f_n\to f$ uniformly. \emph{Continuity of $f$:} for $t_0$ and
		$\varepsilon$, pick $n$ with $d_{\sup}(f,f_n)<\varepsilon/3$, then $\delta$ with $|f_n(t)-f_n(t_0)|<\varepsilon/3$
		for $|t-t_0|<\delta$; then $|f(t)-f(t_0)|\le|f(t)-f_n(t)|+|f_n(t)-f_n(t_0)|+|f_n(t_0)-f(t_0)|<\varepsilon$.
	\end{proof}
	
	\begin{proposition}[Complete $\iff$ closed inside a complete space]\label{prop:complete-closed}
		In a complete metric space, a subspace is complete iff it is closed.
	\end{proposition}
	\begin{proof}
		If $Y$ complete and $x_n\to x$ with $x_n\in Y$, then $(x_n)$ Cauchy in $Y$, so converges in $Y$ to some
		$y\in Y$; uniqueness of limits gives $x=y\in Y$: $Y$ closed. Conversely if $Y$ closed and $(x_n)$ Cauchy
		in $Y$, it converges in $X$ to some $x$; closedness forces $x\in Y$.
	\end{proof}
	
	\begin{theorem}[$\varepsilon$--$\delta$ versus open sets]\label{thm:eps-delta}
		For $f:X\to Y$ between metric spaces, the following are equivalent: (a) $f$ is continuous in the
		open-set sense; (b) $\forall x\ \forall\varepsilon>0\ \exists\delta>0:\ d_X(x',x)<\delta\Rightarrow d_Y(f(x'),f(x))<\varepsilon$.
	\end{theorem}
	\begin{proof}
		(a)$\Rightarrow$(b): $B(f(x),\varepsilon)$ is open, so $f^{-1}(B(f(x),\varepsilon))$ is open and contains $x$;
		hence it contains some $B(x,\delta)$. (b)$\Rightarrow$(a): let $V$ open and $x\in f^{-1}(V)$; then
		$B(f(x),\varepsilon)\subseteq V$ for some $\varepsilon$; (b) gives $\delta$ with
		$f(B(x,\delta))\subseteq B(f(x),\varepsilon)\subseteq V$; thus each point of $f^{-1}(V)$ has a ball inside
		$f^{-1}(V)$: open by Proposition~\ref{prop:basis}.
	\end{proof}

	\begin{definition}[Completion]
		A \textbf{completion} of $(X,d)$ is a complete metric space $(\hat X,\hat d)$ with an isometric
		embedding $\iota:X\to\hat X$ whose image is dense in $\hat X$.
	\end{definition}
	
	\begin{theorem}[Existence of completions]\label{thm:completion}
		Every metric space has a completion.
	\end{theorem}
	\begin{proof}
		Let $\mathcal C$ be the set of Cauchy sequences in $X$; declare $(x_n)\sim(y_n)$ iff $d(x_n,y_n)\to0$
		(an equivalence relation --- P0). Set $\hat X=\mathcal C/{\sim}$ and
		$\hat d([(x_n)],[(y_n)])=\lim_n d(x_n,y_n)$. \emph{The limit exists:}
		$|d(x_m,y_m)-d(x_n,y_n)|\le d(x_m,x_n)+d(y_m,y_n)\to0$, so $(d(x_n,y_n))$ is Cauchy in $\R$, hence
		convergent. \emph{Independent of representatives:} if $(x'_n)\sim(x_n)$ and $(y'_n)\sim(y_n)$ then
		$|d(x'_n,y'_n)-d(x_n,y_n)|\le d(x'_n,x_n)+d(y'_n,y_n)\to0$. \emph{Metric axioms:} symmetry and triangle
		pass to the limit; $\hat d=0\iff(x_n)\sim(y_n)$. \emph{Embedding:} $\iota(x)=[(x,x,\dots)]$ satisfies
		$\hat d(\iota(x),\iota(y))=d(x,y)$. \emph{Density:} for $[(x_n)]$ and $\varepsilon$, choose $N$ with
		$d(x_m,x_n)<\varepsilon$ for $m,n\ge N$; then $\hat d(\iota(x_N),[(x_n)])=\lim_m d(x_N,x_m)\le\varepsilon$.
		\emph{Completeness:} let $([\xi^{(k)}])_k$ be Cauchy in $\hat X$; by density pick $x_k\in X$ with
		$\hat d(\iota(x_k),\xi^{(k)})<1/k$; then $(x_k)$ is Cauchy in $X$ (triangle inequality), and
		$\hat d(\iota(x_k),\xi^{(k)})\to0$ forces $\iota(x_k)\to$ the given Cauchy class; hence $[(x_k)]\in\hat X$
		is the limit.
	\end{proof}
	
	\begin{theorem}[Uniqueness of completions]\label{thm:completion-unique}
		Any two completions $(\hat X_1,\iota_1)$, $(\hat X_2,\iota_2)$ of $(X,d)$ are related by a unique
		isometry $\Phi:\hat X_1\to\hat X_2$ with $\Phi\circ\iota_1=\iota_2$.
	\end{theorem}
	\begin{proof}[Proof idea]
		$\iota_2\circ\iota_1^{-1}:\iota_1(X)\to\iota_2(X)$ is an isometry between dense subspaces; isometries are
		uniformly continuous, and uniformly continuous maps from a dense subspace into a complete space extend
		uniquely continuously; the extensions of $\Phi$ and of its inverse are mutually inverse isometries by
		uniqueness of continuous extensions from dense subspaces.
	\end{proof}
	
	\begin{example}[Completions we will live with]\label{ex:completions}
		\begin{itemize}
			\item The completion of $(\Q,|\cdot|)$ is $\R$ (Cantor's construction is exactly Theorem~\ref{thm:completion}).
			\item The completion of $((0,1),|\cdot|)$ is $[0,1]$.
			\item The completion of $(\Q,d_p)$ is $\Q_p$, and $\hat\Z=\varprojlim \Z/p^n$ appears in S17--S18:
			the same machine, a different metric, a different world.
			\item $(C[0,1],d_{\sup})$ is already complete (Theorem~\ref{thm:sup-complete}); its $L^2$-metric
			completion is $L^2[0,1]$ (mention; not used later).
		\end{itemize}
	\end{example}
	
	\begin{remark}[Ostrowski preview: the census of completions of $\Q$]\label{rem:ostrowski}
		Ostrowski's theorem (S17) says that, up to equivalence, the only non-trivial metrics on $\Q$ are
		$|\cdot|$ and the $d_p$; hence the completions of $\Q$ are exactly $\R$ and the $\Q_p$. The course's
		slogan: each completion adds precisely the points its Cauchy sequences deserved; the adeles (S22)
		then gather all completions into one space.
	\end{remark}

	\section{Compactness: General and Metric}
	% ══════════════════════════════════════════════
	
	\begin{definition}[Compact]
		$X$ is \textbf{compact} if every open cover $\{U_i\}_{i\in I}$ has a finite subcover.
	\end{definition}
	
	\begin{example}\label{ex:compact}
		\begin{itemize}
			\item Finite spaces are compact; $\{1/n:n\ge1\}\cup\{0\}\subseteq\R$ is compact (one open catches $0$ and hence all but finitely many points).
			\item $(0,1)$ is not compact: $(1/n,1)_{n\ge2}$ covers with no finite subcover; $\R$ is not compact: $(-n,n)_{n\ge1}$.
			\item $S^1$ is compact: continuous image of $[0,1]$ under $t\mapsto e^{2\pi it}$.
		\end{itemize}
	\end{example}
	
	\begin{proposition}[Continuous images of compact sets]\label{prop:compact-image}
		The continuous image of a compact space is compact.
	\end{proposition}
	\begin{proof}
		If $\{V_i\}$ covers $f(X)$ then $\{f^{-1}(V_i)\}$ covers $X$; a finite subcover of the latter maps to a finite subcover of the former.
	\end{proof}
	
	\begin{theorem}[Compactness of the unit interval]\label{thm:unit-interval}
		$[0,1]$ with the standard topology is compact.
	\end{theorem}
	\begin{proof}[Proof by bisection]
		Assume an open cover $\mathcal U$ with no finite subcover. Bisect: of the two halves of $[0,1]$, at
		least one cannot be finitely covered; call it $I_1$. Bisect $I_1$, choose a non-finitely-coverable half
		$I_2$; continue, obtaining nested closed intervals $I_n$ with $|I_n|=2^{-n}$, none finitely covered. By
		the nested-interval property, $\bigcap_n I_n=\{s\}$ for a single $s\in[0,1]$. Some $U\in\mathcal U$
		contains $s$, and $U$ open gives $\varepsilon>0$ with $(s-\varepsilon,s+\varepsilon)\cap[0,1]\subseteq U$.
		Choose $n$ with $2^{-n}<\varepsilon$; since $s\in I_n$ and $|I_n|<\varepsilon$, $I_n\subseteq(s-\varepsilon,s+\varepsilon)$,
		so $\{U\}$ finitely covers $I_n$ --- contradiction.
	\end{proof}
	
	\begin{proposition}[Closed inside compact]\label{prop:closed-in-compact}
		A closed subset of a compact space is compact.
	\end{proposition}
	\begin{proof}
		If $K\subseteq X$ closed and $\{U_i\}$ covers $K$, then $\{U_i\}\cup\{X\setminus K\}$ covers $X$; a finite subcover minus $X\setminus K$ finitely covers $K$.
	\end{proof}
	
	\begin{proposition}[Compact inside Hausdorff]\label{prop:compact-in-hausdorff}
		Compact subsets of Hausdorff spaces are closed.
	\end{proposition}
	\begin{proof}
		Proposition~\ref{prop:hausdorff}(b).
	\end{proof}
	
	\begin{lemma}[Tube lemma]\label{lem:tube}
		If $Y$ is compact and $W\subseteq X\times Y$ is open with $\{x\}\times Y\subseteq W$, then there exists an open $U\ni x$ with $U\times Y\subseteq W$.
	\end{lemma}
	\begin{proof}
		For each $y\in Y$, $(x,y)\in W$, so basic opens $U_y\ni x$, $V_y\ni y$ exist with $U_y\times V_y\subseteq W$.
		The $\{V_y\}$ cover $Y$; compactness gives $V_{y_1},\dots,V_{y_k}$ covering $Y$. Let
		$U=U_{y_1}\cap\cdots\cap U_{y_k}$; then for any $(u,v)\in U\times Y$: $v\in V_{y_j}$ for some $j$, so
		$(u,v)\in U_{y_j}\times V_{y_j}\subseteq W$.
	\end{proof}
	
	\begin{proposition}[Tychonoff, finite case]\label{prop:tychonoff-finite}
		If $X$ and $Y$ are compact, then $X\times Y$ is compact.
	\end{proposition}
	\begin{proof}
		Let $\{W_i\}$ cover $X\times Y$. For each $x\in X$, the slice $\{x\}\times Y\cong Y$ is compact, so finitely
		many $W_{i_1},\dots,W_{i_k}$ cover it; by the tube lemma there is an open $U_x\ni x$ with
		$U_x\times Y\subseteq W_{i_1}\cup\cdots\cup W_{i_k}$. The $\{U_x\}$ cover $X$; compactness gives
		$U_{x_1},\dots,U_{x_m}$ covering $X$; the corresponding finite family of $W_i$'s covers $X\times Y$.
	\end{proof}
	
	\begin{theorem}[Heine--Borel]\label{thm:heine-borel}
		$K\subseteq\R^n$ is compact iff $K$ is closed and bounded.
	\end{theorem}
	\begin{proof}
		($\Leftarrow$): $[a,b]\cong[0,1]$ (affine map), hence compact (Theorem~\ref{thm:unit-interval}); boxes
		$[a_1,b_1]\times\cdots\times[a_n,b_n]$ are compact (Proposition~\ref{prop:tychonoff-finite}, inductively);
		a closed subset of a compact is compact (Proposition~\ref{prop:closed-in-compact}), and bounded $K$ lies in some box.
		($\Rightarrow$): unbounded $K$ is covered by $\{B_n(0)\}$ with no finite subcover; if $K$ is not closed,
		pick $x\in\bar K\setminus K$; then $\{X\setminus\bar B_{1/n}(x)\}$ covers $K$ with no finite subcover.
	\end{proof}
	
	\begin{corollary}[Compactness is a homeomorphism invariant]\label{cor:compact-invariant}
		Hence $[0,1]\not\cong(0,1)\not\cong\R$, and the bijection $[0,1)\to S^1$ of Example~\ref{ex:homeo} is not a homeomorphism.
	\end{corollary}
	
	\begin{corollary}[Extreme values]\label{cor:extreme}
		A continuous $f:X\to\R$ on a nonempty compact space attains its maximum and minimum.
	\end{corollary}
	\begin{proof}
		$f(X)$ is compact (Proposition~\ref{prop:compact-image}), hence closed and bounded (Theorem~\ref{thm:heine-borel}); so $\sup f(X),\inf f(X)\in f(X)$.
	\end{proof}
	
	\begin{definition}[Sequential compactness; total boundedness]
		$X$ is \textbf{sequentially compact} if every sequence has a convergent subsequence; \textbf{totally
			bounded} if for every $\varepsilon>0$ finitely many $\varepsilon$-balls cover $X$.
	\end{definition}
	
	\begin{proposition}[The three notions coincide for metric spaces]\label{prop:metric-compact}
		For metric spaces: compact $\Rightarrow$ sequentially compact $\Rightarrow$ complete and totally bounded;
		and complete $+$ totally bounded $\Rightarrow$ compact.
	\end{proposition}
	\begin{proof}[Proof ideas]
		Compact $\Rightarrow$ seq.\ compact: if some sequence had no convergent subsequence, each $x$ would have a
		neighbourhood $U_x$ meeting the sequence finitely often; finitely many $U_x$ cover $X$, yet the sequence
		has infinitely many terms --- contradiction. Seq.\ compact $\Rightarrow$ totally bounded: else some
		$\varepsilon>0$ admits an infinite $\varepsilon$-separated sequence, which has no Cauchy (hence no
		convergent) subsequence. Seq.\ compact $\Rightarrow$ complete: a Cauchy sequence with a convergent
		subsequence converges to the same limit. Complete $+$ totally bounded $\Rightarrow$ compact: from any open
		cover build, via total boundedness and a Lebesgue-number argument, a finite subcover (details: Munkres \S32).
	\end{proof}
	
	\begin{theorem}[Uniform continuity on compact metric spaces]\label{thm:uniform}
		A continuous map from a compact metric space to a metric space is uniformly continuous.
	\end{theorem}
	\begin{proof}
		Given $\varepsilon$, for each $x$ choose $\delta_x$ with $f(B(x,\delta_x))\subseteq B(f(x),\varepsilon/2)$.
		The balls $B(x,\delta_x/2)$ cover $X$; compactness gives finitely many $B(x_i,\delta_{x_i}/2)$; set
		$\delta=\min_i\delta_{x_i}/2>0$. If $d(x,x')<\delta$, pick $i$ with $x\in B(x_i,\delta_{x_i}/2)$; then
		$d(x',x_i)\le d(x',x)+d(x,x_i)<\delta_{x_i}$; hence both $f(x),f(x')$ lie in $B(f(x_i),\varepsilon/2)$,
		so $d_Y(f(x),f(x'))<\varepsilon$.
	\end{proof}

	\section{Connectedness and Path-Connectedness}
	% ══════════════════════════════════════════════
	
	\begin{definition}[Connected]
		$X$ is \textbf{connected} if it cannot be written $X=U\sqcup V$ with $U,V$ nonempty open; equivalently,
		the only clopen subsets are $\varnothing$ and $X$.
	\end{definition}
	
	\begin{proposition}[Continuous image of connected is connected]\label{prop:connected-image}
		If $X$ is connected and $f:X\to Y$ continuous, then $f(X)$ is connected.
	\end{proposition}
	\begin{proof}
		If $f(X)=U\sqcup V$ is a disconnection then $X=f^{-1}(U)\sqcup f^{-1}(V)$ is a disconnection of $X$.
	\end{proof}
	
	\begin{theorem}[Intervals are connected]\label{thm:intervals}
		Every interval $I\subseteq\R$ is connected; in particular $\R$ and $[0,1]$ are connected.
	\end{theorem}
	\begin{proof}
		Suppose $I=U\sqcup V$ with $U,V$ nonempty open in $I$. Pick $a\in U$, $b\in V$; wlog $a<b$; replace $I$
		by $[a,b]\subseteq I$ and $U,V$ by their intersections with $[a,b]$. Let $s=\sup(U\cap[a,b])$; then
		$s\in[a,b]=U\sqcup V$. If $s\in U$: $U$ open gives $\varepsilon>0$ with $[s,s+\varepsilon)\subseteq U$ when
		$s<b$; then $s+\varepsilon/2\in U$ exceeds the supremum --- impossible; and $s=b$ forces $b\in U\cap V$.
		If $s\in V$: $V$ open gives $\varepsilon>0$ with $(s-\varepsilon,s]\subseteq V$; but $s$ being the least
		upper bound of $U\cap[a,b]$ yields $u\in U\cap(s-\varepsilon,s]\subseteq U\cap V$ --- impossible.
	\end{proof}
	
	\begin{corollary}[Intermediate value theorem]\label{cor:IVT}
		If $f:[a,b]\to\R$ is continuous and $f(a)<c<f(b)$ (or $f(b)<c<f(a)$), then some $x\in(a,b)$ has $f(x)=c$.
	\end{corollary}
	\begin{proof}
		$f([a,b])$ is connected (Proposition~\ref{prop:connected-image}); a connected subset of $\R$ is an
		interval (Theorem~\ref{thm:intervals} applied to its complement's disconnection otherwise); hence it
		contains $c$.
	\end{proof}
	
	\begin{example}\label{ex:connected}
		$\Q$ is totally disconnected: for $x<y$ in $\Q$ pick an irrational $t\in(x,y)$; then
		$(-\infty,t)\cap\Q$ and $(t,\infty)\cap\Q$ disconnect any subset containing $x,y$. Contrast: $\R_{>0}$,
		circles, and all intervals are connected.
	\end{example}
	
	\begin{definition}[Path-connected; components]
		$X$ is \textbf{path-connected} if for every $x,y\in X$ there is a continuous $\gamma:[0,1]\to X$ with
		$\gamma(0)=x$, $\gamma(1)=y$. The \textbf{connected components} of $X$ are the maximal connected subsets;
		they are closed, pairwise disjoint, and cover $X$.
	\end{definition}
	
	\begin{example}[Path-connected vs connected]\label{ex:path}
		\begin{itemize}
			\item $\R^n$, open balls, convex sets: path-connected (straight lines); $S^1$: path-connected ($t\mapsto e^{2\pi it}$).
			\item The topologist's sine curve $\{(x,\sin(1/x)):x>0\}\cup\{(0,y):-1\le y\le1\}$: connected but not path-connected.
		\end{itemize}
	\end{example}
	
	\begin{remark}[Preview: $\pi_1$ and homology (P3)]\label{rem:pi1}
		Connectedness is a coarse invariant (``how many pieces''); the fundamental group $\pi_1(X,x_0)$ (loops up
		to homotopy) is finer: $\pi_1(S^1)\cong\Z$, $\pi_1(\R^n)=0$; homology $H_n(X)$ generalizes further. These
		algebraic invariants turn topology into algebra --- the theme of P3, and later of $\pi_1^{\text{\'et}}$ (S35).
	\end{remark}

	\section{Exercises}
	% ══════════════════════════════════════════════
	\begin{exercise}
		Verify that the cofinite collection on an infinite set satisfies (T1)--(T3), and that it is not
		Hausdorff. Which sequences converge, and to how many limits?
	\end{exercise}
	\begin{exercise}
		Show $\{(a,b]:a<b\}$ is a basis for a topology on $\R$ (the lower-limit topology); prove it is strictly
		finer than the standard topology and Hausdorff, and that $[0,1)$ is open in it.
	\end{exercise}
	\begin{exercise}
		Prove that the subspace topology on $A\subseteq X$ is the coarsest topology making the inclusion
		continuous, and that the quotient topology on $Y$ is the finest making $q:X\to Y$ continuous.
	\end{exercise}
	\begin{exercise}
		Show $\{1/n:n\ge1\}\cup\{0\}$ is compact directly from the definition, and that $\{1/n:n\ge1\}$ (without $0$) is not.
	\end{exercise}
	\begin{exercise}
		Prove Proposition~\ref{prop:balls} in full detail, and show $d_1,d_2,d_\infty$ on $\R^n$ all generate the
		standard topology (compare balls: $B_\infty\subseteq B_2\subseteq B_1\subseteq n\,B_\infty$).
	\end{exercise}
	\begin{exercise}
		Prove Proposition~\ref{prop:complete-closed}; deduce that $\Z\subseteq\R$ is complete, and that
		$\Z_p\subseteq\Q_p$ is closed (the latter using the ultrametric inequality --- preview of S19).
	\end{exercise}
	\begin{exercise}
		In $(\Q,|\cdot|)$, prove directly (without invoking $\R$) that $x_1=1$, $x_{n+1}=\tfrac12(x_n+2/x_n)$ is
		Cauchy and has no limit in $\Q$.
	\end{exercise}
	\begin{exercise}
		Carry out Theorem~\ref{thm:completion} for $X=(0,1)$ explicitly: identify $\hat X$ with $[0,1]$ by
		describing the equivalence classes of Cauchy sequences corresponding to $0$ and $1$.
	\end{exercise}
	\begin{exercise}
		Show the product topology on $\R^\N$ (countable product) is strictly coarser than the box topology, by
		exhibiting a box-open set containing no product-basic neighbourhood of the origin (cf.\ Remark~\ref{rem:adeles}).
	\end{exercise}
	\begin{exercise}
		Prove the tube lemma (Lemma~\ref{lem:tube}) and use it to prove Tychonoff for finite products
		(Proposition~\ref{prop:tychonoff-finite}).
	\end{exercise}
	\begin{exercise}
		Prove Theorem~\ref{thm:uniform} again by contradiction using sequential compactness, and deduce that
		every continuous $f:S^1\to\R$ is uniformly continuous.
	\end{exercise}
	\begin{exercise}
		Prove Theorem~\ref{thm:intervals} by a different method: show the only clopen subsets of $\R$ are
		$\varnothing$ and $\R$.
	\end{exercise}
	\begin{exercise}
		Show the topologist's sine curve (Example~\ref{ex:path}) is connected but not path-connected; and prove
		that path-connected always implies connected.
	\end{exercise}
	\begin{exercise}
		Show $\R/\Z\cong S^1$ via $[x]\mapsto e^{2\pi ix}$: verify it is a well-defined continuous bijection, and
		use Proposition~\ref{prop:compact-hausdorff} to conclude it is a homeomorphism.
	\end{exercise}
